## Every wordlist

import FilterTable from '@components/FilterTable.astro';
import wordlists from '@data/wordlists.json';

All {wordlists.length} downloadable wordlists. Word counts are the full list;
the Small/Medium/Large tiers keep the most frequent 50k/150k/300k words.

<FilterTable
	columns={[
		{ label: 'Id' },
		{ label: 'Language' },
		{ label: 'Words', numeric: true },
		{ label: 'Download', numeric: true },
	]}
	rows={wordlists.map((w) => [
		w.id,
		w.name,
		w.words.toLocaleString('en-US'),
		`${(w.gzBytes / 1024 / 1024).toFixed(w.gzBytes < 1024 * 1024 ? 2 : 1)} MB`,
	])}
	extras={wordlists.map((w) => w.english)}
	placeholder="Filter 333 wordlists…"
	caption="Generated from the app's dictionary catalog; sizes are the gzipped downloads."
/>
